std / time

std.time

Time: Duration (a span of time), Instant (for measuring how long something takes), SystemTime (dates and times of day), and sleep.

source: runtime/std/time.loom
time_example.looman example to run
use std.io.print
use std.time.{Duration, ms, secs, measure}

fun main() {
    let d = secs(1).plus(ms(250))
    print("{d.to_string()} is {d.as_millis()} ms")
    let (total, took) = measure(() => (1..=1000).sum())
    print("sum {total}, took less than a second: {took < secs(1)}")
}
